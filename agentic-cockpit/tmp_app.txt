import { useState, useEffect } from 'react';
import { Bot, FileText, Image as ImageIcon, Mic, Settings, Play, CheckCircle2, ChevronRight, Wand2, Users, Star, TrendingUp, BrainCircuit, Video, Database, Activity, Share2, X, Clock } from 'lucide-react';

const fallbackEmployees = [];

export default function App() {
  const [employees, setEmployees] = useState<any[]>([]);
  const [activeTab, setActiveTab] = useState<'team' | 'studio' | 'import' | 'hub'>('team');
  const [activeStep, setActiveStep] = useState(0);
  const [isRunning, setIsRunning] = useState(false);
  const [topic, setTopic] = useState('');
  const [researchScope, setResearchScope] = useState<'kb_only' | 'kb_and_web' | 'web_only'>('kb_and_web');
  const [moduleFilter, setModuleFilter] = useState('all');
  
  const [ytUrl, setYtUrl] = useState('');
  const [isImporting, setIsImporting] = useState(false);
  const [importStep, setImportStep] = useState(0);

  // New Training / HR State
  const [trainingAgent, setTrainingAgent] = useState<any>(null);
  const [feedbackInput, setFeedbackInput] = useState('');
  const [isTraining, setIsTraining] = useState(false);

  useEffect(() => {
    fetch('http://localhost:4000/api/employees')
      .then(res => res.json())
      .then(data => setEmployees(data))
      .catch(console.error);
  }, []);
  
  const [selectedAgentId, setSelectedAgentId] = useState<string | null>(null);
  const [agentLogs, setAgentLogs] = useState<any[]>([]);
  const [finalContent, setFinalContent] = useState<any>(null);
  
  // Content Hub State
  const [contentPieces, setContentPieces] = useState<any[]>([]);

  // Fetch content pieces on tab switch
  useEffect(() => {
    if (activeTab === 'hub') {
      fetch('http://localhost:4000/api/content')
        .then(res => res.json())
        .then(data => setContentPieces(data))
        .catch(console.error);
    }
  }, [activeTab]);

  const approveHook = async (id: string, hookKey: string) => {
    try {
      await fetch(`http://localhost:4000/api/content/${id}/approve`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ selectedHook: hookKey })
      });
      // Refresh
      const res = await fetch('http://localhost:4000/api/content');
      setContentPieces(await res.json());
    } catch (e) {
      console.error(e);
    }
  };

  const submitFeedback = async () => {
    if (!trainingAgent || !feedbackInput.trim()) return;
    setIsTraining(true);
    try {
      const res = await fetch(`http://localhost:4000/api/employees/${trainingAgent.id}/feedback`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ feedback: feedbackInput })
      });
      const updatedEmployee = await res.json();
      setEmployees(prev => prev.map(e => e.id === updatedEmployee.id ? updatedEmployee : e));
      setTrainingAgent(updatedEmployee);
      setFeedbackInput('');
    } catch (e) {
      console.error(e);
    } finally {
      setIsTraining(false);
    }
  };

  const saveEmployeeProfile = async (id: string, updates: any) => {
    try {
      const res = await fetch(`http://localhost:4000/api/employees/${id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(updates)
      });
      const updatedEmployee = await res.json();
      setEmployees(prev => prev.map(e => e.id === updatedEmployee.id ? updatedEmployee : e));
      if (trainingAgent?.id === id) setTrainingAgent(updatedEmployee);
    } catch (e) {
      console.error(e);
    }
  };

  const openAgentProfile = async (id: string) => {
    setSelectedAgentId(id);
    try {
      const res = await fetch(`http://localhost:4000/api/agents/${id}/activity`);
      if(res.ok) {
        const data = await res.json();
        setAgentLogs(data);
      }
    } catch(e) {
      console.error(e);
    }
  };
  const [importResponse, setImportResponse] = useState<any>(null);

  const startPipeline = async () => {
    setIsRunning(true);
    setActiveStep(0);
    const sessionId = `session_${Date.now()}`;

    // Agent execution order matches studioPipeline indices
    const executeAgent = async (agentId: string, stepIdx: number) => {
      setActiveStep(stepIdx);
      try {
        await fetch(`http://localhost:4000/api/studio/${agentId}`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ sessionId, prompt: topic, researchScope, moduleFilter })
        });
      } catch (e) {
        console.error(e);
      }
    };

    // 1. Elena (Research)
    await executeAgent('research', 1);
    // 2. Mia (Strategy)
    await executeAgent('mia', 2);
    // 3. Leo (Script)
    await executeAgent('script', 3);
    // 4. Voice (Mock)
    // await executeAgent('voice', 4);
    // 5. Visuals (Mock)
    // await executeAgent('visuals', 5);
    // 6. Render (Mock)
    // await executeAgent('render', 6);
    // 7. QA (Sarah)
    await executeAgent('qa', 7);
    // 8. Dist (Mock)
    // await executeAgent('dist', 8);

    // After Phase 1 (QA), we stop the pipeline and tell the user to go to the Content Hub
    try {
      const res = await fetch(`http://localhost:4000/api/studio/session/${sessionId}`);
      if(res.ok) {
        const data = await res.json();
        setFinalContent(data);
      }
    } catch (e) {
      console.error(e);
    }

    setIsRunning(false);
    alert("Phase 1 abgeschlossen! Skript wurde zur Freigabe in den Content Hub verschoben.");
    setActiveTab('hub');
  };

  const startImport = async () => {
    if(!ytUrl) return;
    setIsImporting(true);
    setImportStep(1);
    setImportResponse(null);

    try {
      // Update step visual instantly to indicate backend processing
      const timer = setInterval(() => {
         setImportStep(s => s < 2 ? s + 1 : s);
      }, 1500);

      const response = await fetch("http://localhost:4000/api/ingest/youtube", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ url: ytUrl })
      });
      
      clearInterval(timer);
      
      if (response.ok) {
        const data = await response.json();
        setImportResponse(data);
        setImportStep(3);
      } else {
        alert("Fehler beim Importieren!");
        setImportStep(0);
      }
    } catch (e) {
      alert("Backend nicht erreichbar! Läuft der Node-Server auf Port 4000?");
      setImportStep(0);
    } finally {
      setTimeout(() => setIsImporting(false), 2000);
    }
  };

  // Filter studio agents (exclude Ingest & Analytics for the active render pipeline, include render step)
  const studioPipeline = [
    employees.find(e => e.id === 'research'),
    employees.find(e => e.id === 'mia'),
    employees.find(e => e.id === 'script'),
    employees.find(e => e.id === 'voice'),
    employees.find(e => e.id === 'visuals'),
    { id: 'render', name: 'HyperFrames', avatar: '🎬', role: 'System' },
    employees.find(e => e.id === 'qa'),
    employees.find(e => e.id === 'dist')
  ];

  return (
    <div className="flex h-screen bg-slate-900 text-slate-200 overflow-hidden font-sans">
      {/* Sidebar */}
      <div className="w-72 bg-slate-950 border-r border-slate-800 flex flex-col">
        <div className="p-6">
          <div className="flex items-center gap-3 mb-2">
            <Wand2 className="w-8 h-8 text-indigo-400" />
            <h1 className="text-xl font-bold bg-clip-text text-transparent bg-gradient-to-r from-indigo-400 to-purple-400">
              Creator CEO
            </h1>
          </div>
          <p className="text-xs text-slate-500 uppercase tracking-widest font-bold ml-11">Dashboard</p>
        </div>
        
        <div className="px-4 space-y-2 mt-4">
          <button onClick={() => setActiveTab('team')} className={`w-full flex items-center gap-3 p-3 rounded-xl transition-all ${activeTab === 'team' ? 'bg-indigo-500/10 text-indigo-400' : 'text-slate-400 hover:bg-slate-800/50 hover:text-slate-200'}`}>
            <Users className="w-5 h-5" />
            <span className="font-medium">Mitarbeiter Portal ({employees.length})</span>
          </button>
          <button onClick={() => setActiveTab('hub')} className={`w-full flex items-center gap-3 p-3 rounded-xl transition-all ${activeTab === 'hub' ? 'bg-emerald-500/10 text-emerald-400' : 'text-slate-400 hover:bg-slate-800/50 hover:text-slate-200'}`}>
            <Database className="w-5 h-5" />
            <span className="font-medium">Content Hub</span>
          </button>
          <button onClick={() => setActiveTab('import')} className={`w-full flex items-center gap-3 p-3 rounded-xl transition-all ${activeTab === 'import' ? 'bg-red-500/10 text-red-400' : 'text-slate-400 hover:bg-slate-800/50 hover:text-slate-200'}`}>
            <Video className="w-5 h-5" />
            <span className="font-medium">Wissens-Import (Felix)</span>
          </button>
          <button onClick={() => setActiveTab('studio')} className={`w-full flex items-center gap-3 p-3 rounded-xl transition-all ${activeTab === 'studio' ? 'bg-indigo-500/10 text-indigo-400' : 'text-slate-400 hover:bg-slate-800/50 hover:text-slate-200'}`}>
            <Play className="w-5 h-5" />
            <span className="font-medium">Reel-Studio E2E</span>
          </button>
        </div>
        
        <div className="mt-auto p-4">
          <div className="bg-slate-900 border border-slate-800 p-4 rounded-xl mb-4">
            <div className="flex items-center gap-2 text-slate-400 mb-2">
              <Activity className="w-4 h-4 text-orange-400" />
              <span className="text-xs font-bold uppercase">Feedback Loop</span>
            </div>
            <p className="text-xs text-slate-500">Oliver wartet auf neue IG Insights des letzten Uploads zur Optimierung der Hooks.</p>
          </div>
        </div>
      </div>

      {/* Main Content Area */}
      <div className="flex-1 overflow-y-auto">
        {activeTab === 'team' && (
          <div className="p-10 max-w-7xl mx-auto">
            <header className="mb-10">
              <h2 className="text-3xl font-bold mb-2 text-white">Dein E2E Agenten-Team</h2>
              <p className="text-slate-400 text-lg">9 Core-Rollen für den wasserdichten Content-Zyklus.</p>
            </header>
            
            {trainingAgent ? (
              <div className="bg-slate-800 border border-slate-700 rounded-2xl p-6 mb-8">
                <div className="flex justify-between items-start mb-6">
                  <div className="flex items-center gap-4">
                    <div className={`w-16 h-16 rounded-xl flex items-center justify-center text-3xl border ${trainingAgent.color}`}>{trainingAgent.avatar}</div>
                    <div>
                      <h3 className="text-2xl font-bold text-white leading-tight">Personalakte: {trainingAgent.name}</h3>
                      <p className="text-sm text-slate-400">{trainingAgent.role} | Lvl {trainingAgent.level}</p>
                    </div>
                  </div>
                  <button onClick={() => setTrainingAgent(null)} className="text-slate-400 hover:text-white">
                    <X className="w-6 h-6" />
                  </button>
                </div>

                <div className="grid grid-cols-2 gap-6">
                  {/* Left Column: Memory & Rules */}
                  <div className="space-y-4">
                    <div className="bg-slate-900 p-4 rounded-xl border border-slate-700">
                      <h4 className="text-sm font-bold text-indigo-400 mb-2">System Prompt</h4>
                      <textarea 
                        className="w-full bg-transparent text-sm text-slate-300 resize-none h-24 focus:outline-none"
                        value={trainingAgent.systemPrompt}
                        onChange={(e) => setTrainingAgent({...trainingAgent, systemPrompt: e.target.value})}
                        onBlur={() => saveEmployeeProfile(trainingAgent.id, { systemPrompt: trainingAgent.systemPrompt })}
                      />
                    </div>
                    
                    <div className="bg-slate-900 p-4 rounded-xl border border-slate-700">
                      <h4 className="text-sm font-bold text-red-400 mb-2">No-Go Liste</h4>
                      <ul className="list-disc pl-4 text-sm text-slate-300 space-y-1">
                        {trainingAgent.noGoList.map((item: string, i: number) => <li key={i}>{item}</li>)}
                        {trainingAgent.noGoList.length === 0 && <li className="text-slate-500 italic">Noch keine No-Gos</li>}
                      </ul>
                    </div>

                    <div className="bg-slate-900 p-4 rounded-xl border border-slate-700">
                      <h4 className="text-sm font-bold text-emerald-400 mb-2">Memory / Learnings</h4>
                      <ul className="list-disc pl-4 text-sm text-slate-300 space-y-1">
                        {trainingAgent.memory.map((item: string, i: number) => <li key={i}>{item}</li>)}
                        {trainingAgent.memory.length === 0 && <li className="text-slate-500 italic">Noch keine Learnings</li>}
                      </ul>
                    </div>
                  </div>

                  {/* Right Column: Training Input */}
                  <div className="bg-indigo-900/20 border border-indigo-500/30 rounded-xl p-5 flex flex-col">
                    <h4 className="font-bold text-indigo-300 mb-2 flex items-center gap-2">
                      <BrainCircuit className="w-4 h-4" /> Feedback & Training
                    </h4>
                    <p className="text-xs text-indigo-200/70 mb-4">
                      Als CEO kannst du {trainingAgent.name} hier Feedback in natürlicher Sprache geben. Die KI wird das Feedback analysieren und als harte Regel in die No-Go Liste oder das Memory übernehmen.
                    </p>
                    <textarea 
                      placeholder="z.B. Die Hooks im letzten Video waren zu lang. Bitte Hooks künftig unter 15 Wörtern halten!"
                      className="w-full bg-slate-900/50 border border-indigo-500/30 rounded-lg p-3 text-sm text-slate-200 focus:outline-none focus:border-indigo-400 flex-1 resize-none mb-4"
                      value={feedbackInput}
                      onChange={(e) => setFeedbackInput(e.target.value)}
                    />
                    <button 
                      onClick={submitFeedback}
                      disabled={isTraining || !feedbackInput.trim()}
                      className="w-full bg-indigo-500 hover:bg-indigo-600 disabled:opacity-50 text-white font-bold py-2 rounded-lg transition"
                    >
                      {isTraining ? 'Analysiert...' : 'Feedback verarbeiten'}
                    </button>
                  </div>
                </div>
              </div>
            ) : (
              <div className="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-3 gap-6">
                {employees.map(emp => (
                  <div key={emp.id} className="bg-slate-800/50 border border-slate-700 rounded-2xl p-5 hover:bg-slate-800 transition shadow-xl flex flex-col">
                    <div className="flex justify-between items-start mb-4">
                      <div className="flex items-center gap-3">
                        <div className={`w-12 h-12 rounded-xl flex items-center justify-center text-2xl border ${emp.color}`}>{emp.avatar}</div>
                        <div>
                          <h3 className="text-lg font-bold text-white leading-tight">{emp.name}</h3>
                          <p className="text-xs text-slate-400">{emp.role}</p>
                        </div>
                      </div>
                    </div>
                    <p className="text-slate-300 text-xs mb-4 leading-relaxed flex-1">{emp.description}</p>
                    <div className="flex flex-wrap gap-1.5 mb-4">
                      {emp.traits.map((t: string) => <span key={t} className="text-[10px] px-2 py-0.5 rounded bg-slate-900 text-slate-300 border border-slate-700">{t}</span>)}
                    </div>
                    <div className="pt-3 border-t border-slate-700/50 flex justify-between items-center mt-auto">
                      <span className="text-xs font-bold text-white">Lvl {emp.level}</span>
                      <button onClick={() => setTrainingAgent(emp)} className="flex items-center gap-1.5 text-indigo-400 hover:text-indigo-300 text-xs font-medium transition cursor-pointer">
                        <BrainCircuit className="w-3.5 h-3.5" /> Trainieren / Akte
                      </button>
                    </div>
                  </div>
                ))}
              </div>
            )}
          </div>
        )}

        {/* YOUTUBE IMPORT */}
        {activeTab === 'import' && (
          <div className="p-10 max-w-4xl mx-auto">
             <header className="mb-10">
              <h2 className="text-3xl font-bold mb-2 text-white">YouTube &rarr; Obsidian</h2>
            </header>
            <div className="bg-slate-800 rounded-2xl border border-slate-700 p-8 shadow-2xl">
              <div className="flex gap-4 items-end mb-10">
                <div className="flex-1">
                  <label className="block text-sm font-bold text-slate-300 mb-3 uppercase tracking-wide">YouTube Video URL</label>
                  <input type="text" placeholder="https://youtube.com/watch?v=..." value={ytUrl} onChange={(e) => setYtUrl(e.target.value)} className="w-full bg-slate-900 border border-slate-600 rounded-xl p-4 text-white font-medium focus:outline-none focus:border-red-500 focus:ring-1 focus:ring-red-500 transition shadow-inner" />
                </div>
                <button onClick={startImport} disabled={isImporting || !ytUrl} className="bg-red-600 hover:bg-red-500 text-white font-bold py-4 px-8 rounded-xl flex items-center gap-2 disabled:opacity-50 transition transform hover:scale-105 active:scale-95 shadow-xl">
                  <Database className="w-5 h-5" /> Import
                </button>
              </div>

              {/* Status UI */}
              <div className="bg-slate-900 rounded-xl p-6 border border-slate-700/50 min-h-[200px]">
                {importStep === 0 && !importResponse && (
                  <div className="h-full flex flex-col items-center justify-center text-slate-500">
                    <Bot className="w-12 h-12 mb-3 opacity-20" />
                    <p>Felix wartet auf einen Auftrag...</p>
                  </div>
                )}
                
                {importStep > 0 && (
                  <div className="space-y-4">
                    <div className={`flex items-center gap-3 transition-opacity ${importStep >= 1 ? 'opacity-100 text-blue-400' : 'opacity-30'}`}>
                      {importStep > 1 ? <CheckCircle2 className="w-5 h-5 text-emerald-500" /> : <div className="w-5 h-5 border-2 border-blue-500 border-t-transparent rounded-full animate-spin"></div>}
                      <span>Sende URL an lokales Backend...</span>
                    </div>
                    <div className={`flex items-center gap-3 transition-opacity ${importStep >= 2 ? 'opacity-100 text-purple-400' : 'opacity-30'}`}>
                      {importStep > 2 ? <CheckCircle2 className="w-5 h-5 text-emerald-500" /> : (importStep === 2 ? <div className="w-5 h-5 border-2 border-purple-500 border-t-transparent rounded-full animate-spin"></div> : <div className="w-5 h-5 rounded-full border-2 border-slate-700"></div>)}
                      <span>Agent Felix zieht Transkript und extrahiert Fakten...</span>
                    </div>
                    <div className={`flex items-center gap-3 transition-opacity ${importStep >= 3 ? 'opacity-100 text-emerald-400' : 'opacity-30'}`}>
                       {importStep >= 3 ? <CheckCircle2 className="w-5 h-5 text-emerald-500" /> : <div className="w-5 h-5 rounded-full border-2 border-slate-700"></div>}
                      <span>Als Markdown in Obsidian gespeichert!</span>
                    </div>
                  </div>
                )}

                {importResponse && (
                  <div className="mt-8 p-4 bg-emerald-900/20 border border-emerald-500/30 rounded-lg">
                    <p className="text-emerald-400 font-bold mb-2">{importResponse.message}</p>
                    <p className="text-slate-300 text-sm">Gespeichert unter: <code className="bg-black/50 px-2 py-1 rounded">{importResponse.path}</code></p>
                  </div>
                )}
              </div>

            </div>
          </div>
        )}

        {/* PRODUCTION STUDIO */}
        {activeTab === 'studio' && (
          <div className="p-10 max-w-7xl mx-auto">
            <header className="mb-10">
              <h2 className="text-3xl font-bold mb-2 text-white">Reel-Studio (E2E Pipeline)</h2>
              <p className="text-slate-400 text-lg">Der wasserdichte Produktionszyklus inkl. A/B-Hooks, Video-QA und Upload-Paket.</p>
            </header>

            <div className="bg-slate-800 rounded-2xl border border-slate-700 p-8 shadow-2xl">
              <div className="mb-10 relative">
                <label className="block text-sm font-bold text-slate-300 mb-3 uppercase tracking-wide flex items-center gap-2">
                  <Wand2 className="w-4 h-4 text-purple-400" /> 
                  Creative Director Briefing
                </label>
                <div className="relative group mb-4">
                  <div className="absolute -inset-0.5 bg-gradient-to-r from-indigo-500 to-purple-500 rounded-xl opacity-20 group-hover:opacity-40 transition duration-500 blur"></div>
                  <textarea 
                    value={topic} 
                    onChange={(e) => setTopic(e.target.value)} 
                    placeholder="Bsp: Sammle Informationen wie sich Entwicklungstrauma heute zeigt, warum wir alle betroffen sind und erstelle ein anschauliches Video dazu..."
                    className="relative w-full bg-slate-900 border border-slate-700 rounded-xl p-5 text-slate-200 font-medium focus:outline-none focus:border-purple-500 focus:ring-1 focus:ring-purple-500 transition shadow-inner min-h-[120px] resize-y placeholder:text-slate-600"
                  />
                </div>
                
                <div className="flex flex-col gap-4">
                  <div className="flex items-center gap-4 text-sm">
                    <span className="text-slate-400 font-medium w-36">Recherche-Fokus:</span>
                    <label className="flex items-center gap-2 cursor-pointer">
                      <input 
                        type="radio" 
                        name="scope" 
                        value="kb_and_web" 
                        checked={researchScope === 'kb_and_web'} 
                        onChange={(e) => setResearchScope(e.target.value as any)}
                        className="accent-purple-500"
                      />
                      <span className={researchScope === 'kb_and_web' ? 'text-purple-400 font-bold' : 'text-slate-400'}>KB + Internet</span>
                    </label>
                    <label className="flex items-center gap-2 cursor-pointer">
                      <input 
                        type="radio" 
                        name="scope" 
                        value="kb_only" 
                        checked={researchScope === 'kb_only'} 
                        onChange={(e) => setResearchScope(e.target.value as any)}
                        className="accent-indigo-500"
                      />
                      <span className={researchScope === 'kb_only' ? 'text-indigo-400 font-bold' : 'text-slate-400'}>Nur Knowledge Base</span>
                    </label>
                    <label className="flex items-center gap-2 cursor-pointer">
                      <input 
                        type="radio" 
                        name="scope" 
                        value="web_only" 
                        checked={researchScope === 'web_only'} 
                        onChange={(e) => setResearchScope(e.target.value as any)}
                        className="accent-emerald-500"
                      />
                      <span className={researchScope === 'web_only' ? 'text-emerald-400 font-bold' : 'text-slate-400'}>Nur Internet</span>
                    </label>
                  </div>
                  
                  <div className={`flex items-center gap-4 text-sm transition-opacity ${researchScope === 'web_only' ? 'opacity-30 pointer-events-none' : 'opacity-100'}`}>
                    <span className="text-slate-400 font-medium w-36">Wissens-Quelle:</span>
                    <select 
                      value={moduleFilter} 
                      onChange={(e) => setModuleFilter(e.target.value)}
                      className="bg-slate-900 border border-slate-700 rounded-lg px-3 py-2 text-slate-300 focus:outline-none focus:border-indigo-500 focus:ring-1 focus:ring-indigo-500 text-sm"
                    >
                      <option value="all">Gesamte Knowledge Base (Auto-Search)</option>
                      <option value="1_Die_Befreiung">Modul 1: Die Befreiung (Lektion 1-14)</option>
                      <option value="2_Somatic_Developmental_Trauma">Modul 2: Somatic Developmental Trauma</option>
                      <option value="3_Embodied_Relationship_Intimacy">Modul 3: Embodied Relationship & Intimacy</option>
                      <option value="4_Neo_Emotional_Release">Modul 4: Neo Emotional Release</option>
                      <option value="5_The_Awakened_Masculine">Modul 5: The Awakened Masculine</option>
                      <option value="YouTube_Imports">Zusatz: YouTube Import-Skripte (Felix)</option>
                    </select>
                  </div>
                </div>
              </div>

              <div className="bg-slate-900 rounded-xl p-6 mb-10 border border-slate-700/50 shadow-inner">
                {/* Pipeline visual */}
                <div className="flex justify-between items-center mb-8 relative px-2">
                  <div className="absolute left-4 right-4 top-1/2 -translate-y-1/2 h-1 bg-slate-800 rounded-full"></div>
                  <div className="absolute left-4 top-1/2 -translate-y-1/2 h-1 bg-gradient-to-r from-cyan-500 via-purple-500 to-emerald-500 rounded-full transition-all duration-700 ease-in-out" style={{ width: `${Math.min((activeStep / (studioPipeline.length - 1)) * 100, 100)}%`, maxWidth: 'calc(100% - 2rem)' }}></div>
                  
                  {studioPipeline.map((step, idx) => (
                    <div key={idx} className="relative z-10 flex flex-col items-center gap-2">
                      <div className={`w-10 h-10 rounded-full flex items-center justify-center text-lg border-2 shadow-lg transition-all duration-300 ${
                        idx < activeStep ? 'bg-slate-800 border-indigo-500' : 
                        idx === activeStep && isRunning ? 'bg-slate-800 border-purple-400 animate-pulse scale-125' : 
                        'bg-slate-900 border-slate-800 opacity-50 grayscale'
                      }`}>
                        {idx < activeStep ? <CheckCircle2 className="w-5 h-5 text-indigo-400" /> : step?.avatar}
                      </div>
                      <span className={`text-[9px] font-bold uppercase tracking-wider ${idx <= activeStep ? 'text-white' : 'text-slate-500'}`}>{step?.name}</span>
                    </div>
                  ))}
                </div>

                {/* Status Console */}
                <div className="bg-black/40 rounded-lg p-5 min-h-[160px] border border-slate-800 font-mono text-sm leading-relaxed">
                  {activeStep === 0 && !isRunning && <p className="text-slate-500">Warten auf Creative Director Briefing (Dein Prompt)...</p>}
                  {activeStep === 0 && isRunning && <p className="text-cyan-400">» System: Initialisiere RAG Pipeline...</p>}
                  {activeStep === 1 && <p className="text-blue-400">» Elena: Scanne gesamte Knowledge Base via Semantic Search und extrahiere Fakten...</p>}
                  {activeStep === 2 && <p className="text-purple-400">» Mia: Analysiere Prompt-Intent. Passe Content-Format und visuelle Strategie an...</p>}
                  {activeStep === 3 && <p className="text-green-400">» Leo: Schreibe dynamisches Skript/Drehbuch (inkl. 3 A/B Hooks) basierend auf Strategie...</p>}
                  {activeStep === 4 && <p className="text-pink-400">» Arthur: Generiere emotionales Voiceover (ElevenLabs)...</p>}
                  {activeStep === 5 && <p className="text-orange-400">» Maya: Erstelle B-Roll Bilder, Storyboards und kinetische Typografie-Assets...</p>}
                  {activeStep === 6 && <p className="text-amber-400">» System: HyperFrames generiert fertige Video-Versionen...</p>}
                  {activeStep === 7 && <p className="text-teal-400">» Sarah: Multimodal QA! Prüfe Videomaterial und Skript. Sende an Content Hub zur Freigabe.</p>}
                  
                  {activeStep >= 8 && (
                    <div className="text-emerald-400 mt-2 bg-emerald-900/20 p-4 rounded-lg border border-emerald-500/30">
                      <p className="font-bold flex items-center gap-2 text-base mb-2">
                        <CheckCircle2 className="w-6 h-6" /> Reel-Paket 100% fertig für deinen Upload!
                      </p>
                      <ul className="text-slate-300 text-xs space-y-1 list-disc ml-6">
                        <li>3 Video-Versionen (mp4) exportiert (Für Testing)</li>
                        <li>Caption & Hashtags in Zwischenablage bereit</li>
                        <li>ManyChat Call-to-Action verifiziert</li>
                      </ul>
                      <p className="text-orange-400 text-xs mt-3">⏳ Oliver wartet auf Insights 24h nach deinem Upload.</p>
                    </div>
                  )}
                </div>
              </div>

              <div className="flex justify-end mb-6">
                <button onClick={startPipeline} disabled={isRunning || activeStep >= 9} className="bg-indigo-600 hover:bg-indigo-500 text-white font-bold py-4 px-8 rounded-xl flex items-center gap-3 transition shadow-xl disabled:opacity-50">
                  {isRunning ? 'Pipeline läuft...' : 'E2E Produktion starten'} 
                </button>
              </div>


            </div>
          </div>
        )}

        {/* CONTENT HUB */}
        {activeTab === 'hub' && (
          <div className="p-10 max-w-7xl mx-auto">
            <header className="mb-10 flex justify-between items-center">
              <div>
                <h2 className="text-3xl font-bold mb-2 text-white">Content Hub</h2>
                <p className="text-slate-400 text-lg">Verwalte generierte Skripte, wähle Hooks und gib Produktionen frei.</p>
              </div>
              <button onClick={() => setActiveTab('studio')} className="bg-indigo-600 hover:bg-indigo-500 text-white font-bold py-2 px-6 rounded-xl transition shadow-xl">
                + Neues Reel
              </button>
            </header>

            <div className="grid grid-cols-1 gap-6">
              {contentPieces.length === 0 ? (
                <div className="text-center p-12 bg-slate-800/50 rounded-2xl border border-slate-700">
                  <Database className="w-12 h-12 text-slate-600 mx-auto mb-4" />
                  <p className="text-slate-400 text-lg">Noch keine Content-Pieces vorhanden.</p>
                  <p className="text-slate-500 text-sm mt-2">Gehe ins Reel-Studio, um dein erstes Skript zu generieren.</p>
                </div>
              ) : (
                contentPieces.map(piece => (
                  <div key={piece.id} className="bg-slate-800 rounded-2xl border border-slate-700 p-6 shadow-xl relative overflow-hidden">
                    {/* Status Badge */}
                    <div className="absolute top-6 right-6">
                      <span className={`px-3 py-1 rounded-full text-xs font-bold ${piece.status === 'review' ? 'bg-amber-500/20 text-amber-400 border border-amber-500/30' : piece.status === 'production' ? 'bg-emerald-500/20 text-emerald-400 border border-emerald-500/30' : 'bg-slate-700 text-slate-300'}`}>
                        {piece.status === 'review' ? 'Wartet auf Freigabe' : piece.status === 'production' ? 'In Produktion 🎬' : piece.status}
                      </span>
                    </div>

                    <div className="flex items-center gap-3 mb-4">
                      <FileText className="w-6 h-6 text-indigo-400" />
                      <h3 className="text-xl font-bold text-white">{piece.topic}</h3>
                    </div>

                    <div className="grid grid-cols-1 md:grid-cols-2 gap-8 mt-6">
                      <div>
                        <h4 className="text-slate-400 font-bold mb-3 uppercase tracking-wider text-xs">Hooks (Wähle einen aus)</h4>
                        <div className="space-y-3">
                          {['hook_1', 'hook_2', 'hook_3'].map((h, i) => {
                            const hookText = piece.qaData?.final_script?.[h] || piece.scriptData?.[h];
                            const isSelected = piece.selectedHook === h;
                            if (!hookText) return null;
                            return (
                              <div 
                                key={h} 
                                onClick={() => piece.status === 'review' && approveHook(piece.id, h)}
                                className={`p-4 rounded-xl border transition cursor-pointer ${isSelected ? 'bg-indigo-900/40 border-indigo-500' : 'bg-slate-900/50 border-slate-700 hover:border-slate-500'}`}
                              >
                                <div className="flex justify-between items-center mb-2">
                                  <span className={`text-xs font-bold ${isSelected ? 'text-indigo-400' : 'text-slate-400'}`}>Option {i+1}</span>
                                  {isSelected && <CheckCircle2 className="w-4 h-4 text-indigo-400" />}
                                </div>
                                <p className="text-slate-200 text-sm leading-relaxed">{hookText}</p>
                              </div>
                            );
                          })}
                        </div>
                      </div>

                      <div>
                        <h4 className="text-slate-400 font-bold mb-3 uppercase tracking-wider text-xs">Body</h4>
                        <div className="bg-slate-900/50 p-4 rounded-xl border border-slate-700 h-full">
                          <p className="text-slate-200 text-sm leading-relaxed whitespace-pre-wrap">
                            {piece.qaData?.final_script?.body || piece.scriptData?.body || 'Noch kein Body generiert.'}
                          </p>
                        </div>
                      </div>
                    </div>

                    {piece.status === 'review' && !piece.selectedHook && (
                      <div className="mt-6 p-4 bg-amber-900/20 border border-amber-500/30 rounded-xl flex items-center justify-between">
                        <p className="text-amber-400 text-sm font-medium">Klicke oben auf einen Hook, um dieses Skript für die Video-Produktion freizugeben.</p>
                      </div>
                    )}
                  </div>
                ))
              )}
            </div>
          </div>
        )}
      </div>

      {/* AGENT PROFILE MODAL */}
      {selectedAgentId && (
        <div className="fixed inset-0 bg-black/80 backdrop-blur-sm z-50 flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-700 rounded-2xl w-full max-w-2xl max-h-[85vh] flex flex-col shadow-2xl overflow-hidden">
            <div className="p-6 border-b border-slate-800 flex justify-between items-center bg-slate-800/30">
              <div className="flex items-center gap-4">
                <div className={`w-12 h-12 rounded-xl flex items-center justify-center text-2xl border ${employees.find(e => e.id === selectedAgentId)?.color}`}>
                  {employees.find(e => e.id === selectedAgentId)?.avatar}
                </div>
                <div>
                  <h3 className="text-xl font-bold text-white">{employees.find(e => e.id === selectedAgentId)?.name}</h3>
                  <p className="text-sm text-slate-400">{employees.find(e => e.id === selectedAgentId)?.role}</p>
                </div>
              </div>
              <button onClick={() => setSelectedAgentId(null)} className="text-slate-400 hover:text-white transition">
                <X className="w-6 h-6" />
              </button>
            </div>
            
            <div className="p-6 overflow-y-auto flex-1">
              <h4 className="text-slate-300 font-bold mb-4 flex items-center gap-2 uppercase text-xs tracking-wider">
                <Activity className="w-4 h-4 text-indigo-400" /> Activity Log
              </h4>
              
              {agentLogs.length === 0 ? (
                <div className="text-center py-10 bg-slate-800/50 rounded-xl border border-slate-700/50">
                  <Clock className="w-8 h-8 text-slate-600 mx-auto mb-3" />
                  <p className="text-slate-400">Noch keine Tasks ausgeführt.</p>
                </div>
              ) : (
                <div className="space-y-4">
                  {agentLogs.map((log) => (
                    <div key={log.id} className="bg-slate-800/80 border border-slate-700 rounded-xl p-4 flex flex-col gap-2">
                      <div className="flex justify-between items-start">
                        <div className="flex items-center gap-2">
                          {log.status === 'success' ? (
                            <CheckCircle2 className="w-4 h-4 text-green-400" />
                          ) : (
                            <div className="w-4 h-4 rounded-full bg-red-500/20 flex items-center justify-center"><div className="w-2 h-2 rounded-full bg-red-500"></div></div>
                          )}
                          <span className="font-semibold text-white text-sm">{log.action}</span>
                        </div>
                        <span className="text-[10px] text-slate-500 font-mono">
                          {new Date(log.timestamp).toLocaleTimeString('de-DE')}
                        </span>
                      </div>
                      <p className="text-xs text-slate-300 leading-relaxed pl-6">{log.details}</p>
                      {log.model && (
                        <div className="pl-6 pt-2 flex">
                          <span className="bg-slate-900 border border-slate-700 text-[10px] px-2 py-0.5 rounded text-slate-400">
                            Model: {log.model}
                          </span>
                        </div>
                      )}
                    </div>
                  ))}
                </div>
              )}
            </div>
          </div>
        </div>
      )}

    </div>
  );
}
